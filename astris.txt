local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer

local executorName = "Unknown"

pcall(function()
    if identifyexecutor then
        local name = identifyexecutor()
        if type(name) == "string" then
            executorName = name
        end
    elseif getexecutorname then
        local name = getexecutorname()
        if type(name) == "string" then
            executorName = name
        end
    end
end)

local executor = executorName:lower()

if executor:find("xeno", 1, true) or executor:find("solara", 1, true) then
    LocalPlayer:Kick("Fuck off external user")
    return
end

    },

    return
end

    return loadstring(game:HttpGet(gameData.url))()
end)

    warn("Failed to load " .. gameData.name .. ": " .. tostring(result))
end
